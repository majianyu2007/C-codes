题目描述

一张扑克牌的牌面由花色（suit）和点数（rank）组成。其中，花色一共4种，分别为S（黑桃）、H（红桃）、C（梅花）、D（方块）；点数一共13级，不含大小王，由低到高分别为 2、3、4、5、6、7、8、9、T、J、Q、K、A。其中，T-A对应的点数值分别为10-14。要求从键盘输入一手牌共5张，判断其牌型并计算5张牌的点数值之和。牌型和点数和用如下结构体存储： 
typedef struct{
	int type; // 牌型
	int rank; // 点数值之和
}TR, *pTR;
请用wget从http://202.117.179.201/attach/poker.c下载程序主体框架poker.c（其中包括头文件、结构体定义、宏常量、函数原型、main函数、read_hand函数体），编写并提交原型为:
   void judge_hand(pTR result, const int num_in_rank[], const int num_in_suit[]);

的函数。该函数的功能是基于形参所提供的一手牌的统计信息判别牌型并计算点数和，最后将结果存入result所指向的结构体。判别规则和例子如下表：

输入的一手牌 result->rank result->type 牌型英文名 规则
H2 H4 H8 H9 H7 30 0 Flush 5张牌的花色都相同，但不是顺子
SA DT DK CJ SQ 60 1 Straight 5张牌可排成一个点数依次递增1的序列，但不是同花
S3 S4 S6 S5 S7 25 2 Flush-Straight 既是同花、又是顺子
S3 S4 S6 S7 D8 28 3 Others 非同花、非顺子


重要提示：

  1. 输入一手牌的函数为read_hand，已在poker.c中给出。键盘输入时，先输入花色字符，再输入点数字符，两张牌之间以空白符分隔，按回车键结束输入。例如，H3 C9 DJ D6 SK表示输入的5张牌分别为红桃3、梅花9、方块J、方块6、黑桃K。另外，该函数统计各个点数、各种花色的牌的张数，分别存入num_in_rank和num_in_suit数组。例如，num_in_rank[0]保存点数为2的牌的张数，num_in_suit[0]保存花色为黑桃的牌的张数。
  2. 同花判别算法：num_in_suit数组中，任一元素的值为5，则说明是同花；顺子判别算法：从num_in_rank数组的第一个非0元素开始，统计连续出现的非0元素个数，当遇到下一个0元素则停止统计。统计结束时，如非0元素的个数等于5则是顺子（可参阅教材P180页例题）。
  3. 在编写judge_hand函数时，可直接使用poker.c中所定义的宏。
  4. 仅需提交judge_hand函数。可用sed -n “L1,L2p” xxx.c | xclip命令从xxx.c文件中先复制L1至L2行的代码，再Ctrl+V粘贴至网页提交。这里，L1、L2分别表示judge_hand函数的起始行号和结束行号。

      5. #include <stdio.h>

#define NUM_RANKS 13
#define NUM_SUITS 4
#define NUM_CARDS 5

typedef struct{
    int type;
    int rank;
}TR, *pTR;

int read_hand(int num_in_rank[], int num_in_suit[]);
void judge_hand(pTR result, const int num_in_rank[], const int num_in_suit[]);

int main(void)
{
    int num_in_rank[NUM_RANKS];
    int num_in_suit[NUM_SUITS];

    TR result;

    if (!read_hand(num_in_rank, num_in_suit))
    {
        puts("Failed to read one hand");
        return 0; //由于OJ系统需要，返回0值
    }
    else
        puts("Read one hand ok");

    judge_hand(&result, num_in_rank, num_in_suit);
    switch (result.type)
    {
        case 0:
              printf("\nFlush %d\n", result.rank); break;
        case 1:
              printf("\nStraight %d\n", result.rank); break;
        case 2:
              printf("\nFlush-Straight %d\n", result.rank); break;
        default:
              printf("\nOthers %d\n", result.rank);
      }

      return 0;
}

int read_hand(int num_in_rank[], int num_in_suit[])
{
    char rank_ch, suit_ch;
    int cards_read = 0;
    int rank, suit;
    int readed[NUM_SUITS][NUM_RANKS] = {0};

    for (rank = 0; rank < NUM_RANKS; rank++)
        num_in_rank[rank] = 0;

    for (suit = 0; suit < NUM_SUITS; suit++)
        num_in_suit[suit] = 0;

    puts("Please input one hand of five cards:");
    while (cards_read < NUM_CARDS)
    {
        suit_ch = getchar();
        switch(suit_ch)
        {
             case 's': case 'S': suit = 0; break;
             case 'h': case 'H': suit = 1; break;
             case 'c': case 'C': suit = 2; break;
             case 'd': case 'D': suit = 3; break;
             default: suit = -1;
         }
         rank_ch = getchar();
         switch(rank_ch)
         {
             case '2': rank = 0; break;
             case '3': rank = 1; break;
             case '4': rank = 2; break;
             case '5': rank = 3; break;
             case '6': rank = 4; break;
             case '7': rank = 5; break;
             case '8': rank = 6; break;
             case '9': rank = 7; break;
             case 't': case 'T': rank = 8; break;
             case 'j': case 'J': rank = 9; break;
             case 'q': case 'Q': rank = 10; break;
             case 'k': case 'K': rank = 11; break;
             case 'a': case 'A': rank = 12; break;
             default: rank = -1;
         }

         getchar();
         if (suit == -1 || rank == -1)
         {
             printf("%c%c is invalid\n", suit_ch, rank_ch);
             break;
         }

        if (readed[suit][rank])
        {
           printf("%c%c is already existed\n", suit_ch, rank_ch);
           break;
        }

        num_in_suit[suit]++;
        num_in_rank[rank]++;
        readed[suit][rank] = 1;
        cards_read++;
      }
    
      if (cards_read == NUM_CARDS)
            return 1; // succeeded
      else
            return 0; // failed
}
