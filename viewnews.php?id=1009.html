题目1：成绩统计 

#include <stdio.h>
#include <math.h>

int main(void)
{
    int cnt;
    double score, sum;
    double max = 0.0, min = 100.0;
    double avg;

    cnt = 0;
    sum = 0.0;
    while(1)
    {
        scanf("%lf", &score);
        if(fabs(score + 1) < 1E-10)
            break;
        
        max = (max < score)? score: max;
        min = (min > score)? score: min;

        sum += score;
        cnt++;
    }

    sum -= max + min;
    cnt -= 2;
    avg = sum / cnt;

    printf("%.2f\n", avg);
    return 0;
}

题目2：成绩计算与排序 

void total(int s[], int n, int m)
{
    int i, j;
    int grade;
    int sum;

    for(i = 0; i < n; i++)
    {
        sum = 0;
        for(j = 0; j < m; j++)
        {
            scanf("%d", &grade);
            sum += grade;
        }

        s[i] = sum;
    }
}

int cmp(const void *p, const void *q)
{
    return (*(int *)q - *(int *)q);
}

void sort(int s[],int n)
{
    qsort(s, n, sizeof(s[0]), cmp);
}

题目3：基于BMI的人体体重评价 

char evaBMI(double w, double h)
{
    double b;

    if(!(w > 0) || !(h > 0))
        return 'E';

    b = w / (h * h);

    if(b < 18.5)
        return 'S';
    else if(b < 24.0)
        return 'N';
    else if(b < 28.0)
        return 'O';
    else
        return 'F';
}

题目4：幼儿园分享糖果 

void share(int a[], int n)
{
    int i;
    for (i = 0; i < n; i++) {
        if (i == 0)
            shareCandy(a, a + 1, a + n - 1);
        else if (i == n - 1)
            shareCandy(a + n - 1, a + n - 2, a);
        else
            shareCandy(a + i, a + i - 1, a + i + 1);
    }
}
void shareCandy(int *x, int *y, int *z)
{
    *x /= 3;
    *y += *x;
    *z += *x;
}

题目5：密码合法性检验 

int isValidPasswd (const char *pw, int n)
{
    int i;
    int flag1 = 0, flag2 = 0, flag3 = 0, flag4 = 0;

    for(i = 0; i < n; i++)
    {
        if(pw[i] >= 'a' && pw[i] <= 'z')
            flag1 = 1;
        else if(pw[i] >= 'A' && pw[i] <= 'Z')
            flag2 = 2;
        else if(pw[i] >= '0' && pw[i] <= '9')
            flag3 = 4;
        else
            flag4 = 8;
    }
    return 15 - (flag1 + flag2 + flag3 + flag4);
}
